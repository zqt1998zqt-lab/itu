if(ITU_ORT_ROOT)
  # ITU_ORT_ROOT is an explicit SDK for the target; Android's sysroot must not
  # prepend itself to this path or hide the supplied arm64 shared library.
  find_library(ITU_ORT_LIBRARY onnxruntime PATHS "${ITU_ORT_ROOT}/lib"
    NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
  target_include_directories(itu_audio PRIVATE "${ITU_ORT_ROOT}/include")
  target_link_libraries(itu_audio PRIVATE "${ITU_ORT_LIBRARY}")
  target_compile_definitions(itu_audio PRIVATE ITU_HAS_ORT=1)
endif()
